cmake_minimum_required(VERSION 3.25)
project(CommunityDiversHellForgeLoader LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The standalone HellForge loader requires Windows x64.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(FetchContent)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/upstream.lock.json" CD_HF_LOCK)
foreach(dep upstream minhook)
  string(JSON url GET "${CD_HF_LOCK}" ${dep} url)
  string(JSON sha GET "${CD_HF_LOCK}" ${dep} sha256)
  FetchContent_Declare(${dep} URL "${url}" URL_HASH "SHA256=${sha}"
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR __no_upstream_build)
endforeach()
FetchContent_MakeAvailable(upstream minhook)
set(HF_NATIVE "${upstream_SOURCE_DIR}/native")
set(HF_SDK "${HF_NATIVE}/mod-sdk")
set(MINHOOK_SOURCES
  "${minhook_SOURCE_DIR}/src/buffer.c" "${minhook_SOURCE_DIR}/src/hook.c"
  "${minhook_SOURCE_DIR}/src/trampoline.c" "${minhook_SOURCE_DIR}/src/hde/hde64.c")
function(loader_compile target)
  target_include_directories(${target} PRIVATE "${HF_SDK}"
    "${minhook_SOURCE_DIR}/include" "${minhook_SOURCE_DIR}/src")
  target_compile_options(${target} PRIVATE -Wall -Werror -Wno-unused-parameter)
endfunction()
add_library(hellforge_modloader SHARED "${HF_NATIVE}/mod-loader/hellforge_modloader.c" ${MINHOOK_SOURCES})
set_target_properties(hellforge_modloader PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
loader_compile(hellforge_modloader)
include(CTest)
if(BUILD_TESTING)
  add_executable(hellforge_stats_test "${HF_NATIVE}/mod-loader/test/stats_test.c" ${MINHOOK_SOURCES})
  loader_compile(hellforge_stats_test)
  target_compile_options(hellforge_stats_test PRIVATE -O1)
  add_test(NAME hellforge_synthetic_stats COMMAND hellforge_stats_test)
  set_tests_properties(hellforge_synthetic_stats PROPERTIES TIMEOUT 60)
  add_library(fake_game SHARED "${HF_NATIVE}/mod-loader/test/fake_game.c")
  set_target_properties(fake_game PROPERTIES PREFIX "")
  target_compile_options(fake_game PRIVATE -O1)
  add_library(hook_test_mod SHARED "${HF_NATIVE}/mod-loader/test/hook_test_mod.c")
  set_target_properties(hook_test_mod PROPERTIES PREFIX "")
  loader_compile(hook_test_mod)
  add_executable(loader_smoke tests/loader_smoke.c)
  target_compile_options(loader_smoke PRIVATE -Wall -Wextra -Werror)
  target_link_options(loader_smoke PRIVATE -municode)
  add_test(NAME hellforge_loader_smoke COMMAND loader_smoke
    "$<TARGET_FILE:hellforge_modloader>" "$<TARGET_FILE:fake_game>" "$<TARGET_FILE:hook_test_mod>"
    "${CMAKE_BINARY_DIR}/smoke")
  set_tests_properties(hellforge_loader_smoke PROPERTIES TIMEOUT 20)
endif()
# This component is deliberately not installed by the launcher packaging rules.
# Building it does not register an adapter or enable the current-client launch path.
foreach(name LICENSE LICENSE-EXCEPTION LICENSE-SDK)
  configure_file("${upstream_SOURCE_DIR}/${name}" "${CMAKE_BINARY_DIR}/licenses/HellForge-${name}" COPYONLY)
endforeach()
configure_file("${minhook_SOURCE_DIR}/LICENSE.txt" "${CMAKE_BINARY_DIR}/licenses/MinHook-LICENSE.txt" COPYONLY)
file(GLOB_RECURSE inputs LIST_DIRECTORIES FALSE CONFIGURE_DEPENDS
  "${HF_NATIVE}/mod-loader/*.c" "${HF_NATIVE}/mod-loader/*.h"
  "${HF_SDK}/*.h" "${HF_NATIVE}/common/*.h")
list(SORT inputs)
set(source_digest "")
foreach(path IN LISTS inputs)
  file(SHA256 "${path}" hash)
  file(RELATIVE_PATH name "${upstream_SOURCE_DIR}" "${path}")
  string(APPEND source_digest "${name}:${hash}\n")
endforeach()
string(SHA256 CD_HF_SOURCE_DIGEST "${source_digest}")
string(JSON CD_HF_COMMIT GET "${CD_HF_LOCK}" upstream commit)
string(JSON CD_HF_UPSTREAM_BUILD GET "${CD_HF_LOCK}" upstream game_build)
configure_file(build-info.json.in build-info.json @ONLY)
