cmake_minimum_required(VERSION 3.25)
project(CommunityDivers VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
include(FetchContent)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.12.0
  URL_HASH SHA256=4b92eb0c06d10683f7447ce9406cb97cd4b453be18d7279320f7b2f025c10187 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
add_library(community_core src/core.cpp src/compatibility.cpp)
target_include_directories(community_core PUBLIC include)
target_link_libraries(community_core PUBLIC nlohmann_json::nlohmann_json)
if(WIN32)
  target_compile_definitions(community_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE community_core)
  add_test(NAME core_tests COMMAND core_tests)
endif()
if(WIN32)
  add_library(community_platform src/platform_win.cpp src/identity_win.cpp)
  target_link_libraries(community_platform PUBLIC community_core PRIVATE winhttp shell32 ole32 bcrypt)
  add_executable(CommunityDiversInspect src/inspect_main.cpp)
  target_link_libraries(CommunityDiversInspect PRIVATE community_platform)
  add_custom_target(community_catalog ALL COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_SOURCE_DIR}/compatibility" "$<TARGET_FILE_DIR:CommunityDiversInspect>/compatibility")
  add_dependencies(community_catalog CommunityDiversInspect)
  if(BUILD_TESTING)
    add_executable(platform_tests tests/platform_tests.cpp)
    target_link_libraries(platform_tests PRIVATE community_platform ws2_32)
    add_test(NAME platform_tests COMMAND platform_tests)
    set_tests_properties(platform_tests PROPERTIES TIMEOUT 30)
  endif()
endif()
option(COMMUNITY_BUILD_LAUNCHER "Build the Windows desktop launcher" ${WIN32})
if(COMMUNITY_BUILD_LAUNCHER)
  if(NOT WIN32)
    message(FATAL_ERROR "The desktop launcher currently requires Windows; portable core tests remain available.")
  endif()
  FetchContent_Declare(imgui URL https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.93.0
    URL_HASH SHA256=f7bf6bc76e5d79d40d751096663e6c5d18daf08129dd52e7f7b09e4251c8eeee DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(imgui)
  add_library(imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_compile_definitions(imgui PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(imgui PUBLIC d3d11 dxgi dwmapi)
  if(BUILD_TESTING)
    add_library(imgui_widget_test STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
      ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp)
    target_include_directories(imgui_widget_test PUBLIC ${imgui_SOURCE_DIR})
    target_compile_definitions(imgui_widget_test PUBLIC IMGUI_ENABLE_TEST_ENGINE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_executable(ui_tests tests/ui_tests.cpp src/ui.cpp)
    target_include_directories(ui_tests PRIVATE src)
    target_compile_definitions(ui_tests PRIVATE CD_VERSION="${PROJECT_VERSION}")
    target_link_libraries(ui_tests PRIVATE community_platform imgui_widget_test ws2_32)
    add_test(NAME ui_tests COMMAND ui_tests)
    set_tests_properties(ui_tests PROPERTIES TIMEOUT 30)
  endif()
  enable_language(RC)
  add_executable(CommunityDivers WIN32 src/main.cpp src/ui.cpp src/app.rc src/launcher.manifest)
  add_dependencies(CommunityDivers community_catalog)
  target_link_libraries(CommunityDivers PRIVATE community_platform imgui winhttp shell32 ole32 windowscodecs comdlg32)
  target_compile_definitions(CommunityDivers PRIVATE CD_VERSION="${PROJECT_VERSION}")
  add_custom_command(TARGET CommunityDivers POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:CommunityDivers>/assets")
  install(TARGETS CommunityDivers CommunityDiversInspect RUNTIME DESTINATION .)
  install(DIRECTORY compatibility DESTINATION .)
  install(DIRECTORY assets DESTINATION .)
  install(FILES LICENSE THIRD_PARTY.md README.md PROJECT.md CONTRIBUTING.md DESTINATION .)
  install(DIRECTORY docs DESTINATION .)
  install(FILES ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME imgui.txt)
  install(FILES ${json_SOURCE_DIR}/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json.txt)
endif()
