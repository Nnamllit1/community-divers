cmake_minimum_required(VERSION 3.25)
project(CommunityDivers VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Git QUIET)
set(CD_SOURCE_COMMIT "unknown")
set(CD_SOURCE_DIRTY "true")
set(CD_SOURCE_REVISION "0")
if(GIT_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    OUTPUT_VARIABLE CD_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE CD_GIT_RESULT TIMEOUT 5)
  execute_process(COMMAND "${GIT_EXECUTABLE}" status --porcelain WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    OUTPUT_VARIABLE CD_GIT_STATUS OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE CD_STATUS_RESULT TIMEOUT 5)
  execute_process(COMMAND "${GIT_EXECUTABLE}" rev-list --first-parent --count HEAD WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    OUTPUT_VARIABLE CD_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET TIMEOUT 5)
  execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse --is-shallow-repository WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    OUTPUT_VARIABLE CD_SOURCE_SHALLOW OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET TIMEOUT 5)
  if(CD_SOURCE_SHALLOW STREQUAL "true")
    set(CD_SOURCE_REVISION "0")
  endif()
  if(CD_GIT_RESULT EQUAL 0 AND CD_STATUS_RESULT EQUAL 0 AND CD_GIT_STATUS STREQUAL "")
    set(CD_SOURCE_DIRTY "false")
  endif()
endif()
if(NOT CD_SOURCE_COMMIT MATCHES "^[0-9a-f]+$")
  set(CD_SOURCE_COMMIT "unknown")
endif()
string(SUBSTRING "${CD_SOURCE_COMMIT}" 0 8 CD_SHORT_COMMIT)
set(CD_RELEASE_VERSION "${PROJECT_VERSION}-preview.1")
set(CD_RELEASE_ID "${CD_RELEASE_VERSION}+${CD_SHORT_COMMIT}")
if(NOT CD_SOURCE_REVISION MATCHES "^[0-9]+$")
  set(CD_SOURCE_REVISION "0")
endif()
file(READ "${CMAKE_SOURCE_DIR}/updates/publisher-key.txt" CD_UPDATE_PUBLIC_KEY)
string(STRIP "${CD_UPDATE_PUBLIC_KEY}" CD_UPDATE_PUBLIC_KEY)
string(LENGTH "${CD_UPDATE_PUBLIC_KEY}" CD_UPDATE_PUBLIC_KEY_LENGTH)
if(NOT CD_UPDATE_PUBLIC_KEY_LENGTH EQUAL 144 OR NOT CD_UPDATE_PUBLIC_KEY MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "A valid public update trust key is required.")
endif()
configure_file(cmake/update_trust.hpp.in generated/community/update_trust.hpp @ONLY)
configure_file(cmake/version.hpp.in generated/community/version.hpp @ONLY)
configure_file(cmake/build-info.json.in build-info.json @ONLY)
include(FetchContent)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.12.0
  URL_HASH SHA256=4b92eb0c06d10683f7447ce9406cb97cd4b453be18d7279320f7b2f025c10187 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
add_library(community_core src/core.cpp src/compatibility.cpp src/updates.cpp src/install_inventory.cpp)
target_include_directories(community_core PUBLIC include "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(community_core PUBLIC nlohmann_json::nlohmann_json)
if(WIN32)
  target_compile_definitions(community_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(update_core_tests tests/update_core_tests.cpp)
  target_link_libraries(update_core_tests PRIVATE community_core)
  add_test(NAME update_core_tests COMMAND update_core_tests)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE community_core)
  add_test(NAME core_tests COMMAND core_tests)
endif()

  FetchContent_Declare(miniz URL https://codeload.github.com/richgel999/miniz/tar.gz/refs/tags/3.1.2

    DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR not-added)

  file(WRITE "${CMAKE_BINARY_DIR}/generated/miniz_export.h" "#pragma once\n#define MINIZ_EXPORT\n")

  add_executable(CommunityDiversUpdate src/update_main.cpp)
  target_link_libraries(CommunityDiversUpdate PRIVATE community_platform)
  add_executable(CommunityDiversInspect src/inspect_main.cpp)
  target_link_libraries(CommunityDiversInspect PRIVATE community_platform)

    "${CMAKE_SOURCE_DIR}/compatibility" "$<TARGET_FILE_DIR:CommunityDiversInspect>/compatibility")

  if(BUILD_TESTING)

    target_link_libraries(update_install_fixture PRIVATE community_platform)
    add_executable(update_install_tests tests/update_install_tests.cpp)

    set_tests_properties(update_install_tests PROPERTIES TIMEOUT 90)
    add_executable(update_tests tests/update_tests.cpp)
    target_link_libraries(update_tests PRIVATE community_platform)
    target_compile_definitions(update_tests PRIVATE CD_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
    add_test(NAME update_tests COMMAND update_tests)

    add_executable(platform_tests tests/platform_tests.cpp)
    target_link_libraries(platform_tests PRIVATE community_platform ws2_32)

option(COMMUNITY_BUILD_LAUNCHER "Build the Windows desktop launcher" ${WIN32})
if(COMMUNITY_BUILD_LAUNCHER)
  if(NOT WIN32)
    message(FATAL_ERROR "The desktop launcher currently requires Windows; portable core tests remain available.")
  endif()
  FetchContent_Declare(imgui URL https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.93.0
    URL_HASH SHA256=f7bf6bc76e5d79d40d751096663e6c5d18daf08129dd52e7f7b09e4251c8eeee DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(imgui)
  add_library(imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp)
  target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_compile_definitions(imgui PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(imgui PUBLIC d3d11 dxgi dwmapi)
  if(BUILD_TESTING)
    add_library(imgui_widget_test STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
      ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp)
    target_include_directories(imgui_widget_test PUBLIC ${imgui_SOURCE_DIR})
    target_compile_definitions(imgui_widget_test PUBLIC IMGUI_ENABLE_TEST_ENGINE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_executable(ui_tests tests/ui_tests.cpp src/ui.cpp)
    target_include_directories(ui_tests PRIVATE src)
    target_compile_definitions(ui_tests PRIVATE CD_VERSION="${PROJECT_VERSION}")
    target_link_libraries(ui_tests PRIVATE community_platform imgui_widget_test ws2_32)
    add_test(NAME ui_tests COMMAND ui_tests)
    set_tests_properties(ui_tests PROPERTIES TIMEOUT 30)
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Windows launcher currently targets x64.")
  endif()
  enable_language(RC)
  add_executable(CommunityDivers WIN32 src/main.cpp src/ui.cpp src/app.rc src/launcher.manifest)
  add_dependencies(CommunityDivers community_catalog)
  target_link_libraries(CommunityDivers PRIVATE community_platform imgui winhttp shell32 ole32 windowscodecs comdlg32)
  target_compile_definitions(CommunityDivers PRIVATE CD_VERSION="${PROJECT_VERSION}")
  add_custom_command(TARGET CommunityDivers POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:CommunityDivers>/assets")
  install(TARGETS CommunityDivers CommunityDiversInspect CommunityDiversUpdate RUNTIME DESTINATION .)
  install(FILES updates/preview.json updates/publisher-key.txt DESTINATION updates)
  install(DIRECTORY compatibility DESTINATION .)
  install(FILES "${CMAKE_BINARY_DIR}/build-info.json" DESTINATION .)
  install(DIRECTORY assets DESTINATION .)
  install(FILES LICENSE THIRD_PARTY.md README.md PROJECT.md CONTRIBUTING.md Update.cmd DESTINATION .)
  install(DIRECTORY docs DESTINATION .)
  install(FILES ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME imgui.txt)
  install(FILES ${json_SOURCE_DIR}/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json.txt)
  install(FILES ${miniz_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME miniz.txt)
  install(SCRIPT cmake/install_inventory.cmake)
  set(CPACK_PACKAGE_NAME "CommunityDivers")
  set(CPACK_PACKAGE_VERSION "${CD_RELEASE_VERSION}")
  set(CPACK_PACKAGE_FILE_NAME "CommunityDivers-${CD_RELEASE_VERSION}-${CD_SHORT_COMMIT}-windows-x64")
  set(CPACK_GENERATOR "ZIP")
  set(CPACK_PACKAGE_CHECKSUM "SHA256")
  set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
  include(CPack)
endif()
